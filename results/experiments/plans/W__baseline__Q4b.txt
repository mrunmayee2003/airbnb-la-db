SELECT n.name AS neighbourhood, COUNT(*) AS listings_marked_available
FROM calendar c
JOIN listing l       ON l.listing_id = c.listing_id
JOIN neighbourhood n ON n.neighbourhood_id = l.neighbourhood_id
WHERE c.cal_date = %(date)s
  AND c.available = 1
  AND n.name = %(nbhd)s
GROUP BY n.name
