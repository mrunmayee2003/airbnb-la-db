SELECT h.host_id, h.host_since, h.is_superhost,
       COUNT(l.listing_id)      AS la_listings,
       h.source_listings_count  AS source_listings_count
FROM listing l
JOIN host h ON h.host_id = l.host_id
GROUP BY h.host_id
ORDER BY la_listings DESC
LIMIT %(limit)s
