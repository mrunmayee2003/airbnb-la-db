-- params: {'date': '2027-01-12'}
SELECT /*+ NO_INDEX(c) NO_INDEX(l) NO_INDEX(n) */  n.name AS neighbourhood, COUNT(*) AS listings_marked_available
FROM calendar c
JOIN listing l       ON l.listing_id = c.listing_id
JOIN neighbourhood n ON n.neighbourhood_id = l.neighbourhood_id
WHERE c.cal_date = %(date)s
  AND c.available = 1
GROUP BY n.name
ORDER BY listings_marked_available DESC
