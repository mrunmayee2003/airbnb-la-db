-- P1 | config: INDEX(cal_date) | variant: 7 night window
-- params: {'lo': '2026-10-15', 'hi': '2026-10-21'}
SELECT listing_id, cal_date, price
FROM calendar
WHERE available = 1
  AND price IS NOT NULL
  AND cal_date BETWEEN %(lo)s AND %(hi)s
ORDER BY price
LIMIT 100
